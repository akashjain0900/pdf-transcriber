# Ledger — runtime dependencies.

# PDF rendering. Import as `pymupdf`; the old `fitz` name is deprecated in 1.28+.
pymupdf>=1.24

# HTTP client for the Gemini REST endpoint. Used instead of google-genai so
# that retry, quota and auth behaviour stays under our own control.
requests>=2.31

# Image handling: needed for deskew, and for the local text detection that
# picks the profiler's sample pages.
Pillow>=10.0
numpy>=1.26

# Web UI: the API server and the ASGI server that runs it.
fastapi>=0.110
uvicorn>=0.27

# REQUIRED ON WINDOWS, harmless elsewhere. All quota accounting keys off the US
# Pacific date (when Google resets daily limits), and zoneinfo looks that name
# up in the OS timezone database — which Windows does not have. Without this,
# quota.py fails at import time on Windows.
tzdata; platform_system == 'Windows'
